一、核心算法概述

1.1 二分搜索

二分搜索是"啊哈"时刻的典型代表——简单但极其高效。

// 二分搜索 - 寻找目标值
function binarySearch<T>(arr: T[], target: T, compare: (a: T, b: T) => number): number {
  let left = 0;
  let right = arr.length - 1;

  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    const cmp = compare(arr[mid], target);

    if (cmp === 0) {
      return mid;
    } else if (cmp < 0) {
      left = mid + 1;
    } else {
      right = mid - 1;
    }
  }

  return -1;
}

// 使用示例
const sorted = [1, 3, 5, 7, 9, 11, 13, 15];
console.log(binarySearch(sorted, 7, (a, b) => a - b));  // 3
console.log(binarySearch(sorted, 6, (a, b) => a - b));  // -1

1.2 插入排序

简单但 O(n²) 复杂度，适合小规模数据。

// 插入排序
function insertionSort(arr: number[]): number[] {
  const result = [...arr];  // 复制数组

  for (let i = 1; i < result.length; i++) {
    const key = result[i];
    let j = i - 1;

    // 将比 key 大的元素向后移动
    while (j >= 0 && result[j] > key) {
      result[j + 1] = result[j];
      j--;
    }

    result[j + 1] = key;
  }

  return result;
}

// 测试
console.log(insertionSort([64, 34, 25, 12, 22, 11, 90]));
// [11, 12, 22, 25, 34, 64, 90]

三、经典问题的算法解法

3.1 变位词检测

检测两个字符串是否由相同字符组成（可重排）。

// 方法1：排序后比较 - O(n log n)
function isAnagram1(s1: string, s2: string): boolean {
  const sort = (s: string) => s.split('').sort().join('');
  return sort(s1) === sort(s2);
}

// 方法2：计数 - O(n)
function isAnagram2(s1: string, s2: string): boolean {
  if (s1.length !== s2.length) return false;

  const count: Record<string, number> = {};

  for (const c of s1) {
    count[c] = (count[c] || 0) + 1;
  }

  for (const c of s2) {
    if (!count[c]) return false;
    count[c]--;
  }

  return true;
}

// 测试
console.log(isAnagram1("listen", "silent"));  // true
console.isAnagram2("listen", "silent"));  // true

3.2 数组元素唯一性检测

// 方法1：排序后检查 - O(n log n)
function hasUnique1<T>(arr: T[]): boolean {
  const sorted = [...arr].sort();
  for (let i = 1; i < sorted.length; i++) {
    if (sorted[i] === sorted[i - 1]) return false;
  }
  return true;
}

// 方法2：Set - O(n)
function hasUnique2<T>(arr: T[]): boolean {
  return new Set(arr).size === arr.length;
}

// 方法3：哈希表 - O(n)
function hasUnique3<T>(arr: T[]): boolean {
  const seen = new Set<T>();
  for (const item of arr) {
    if (seen.has(item)) return false;
    seen.add(item);
  }
  return true;
}

// 测试
console.log(hasUnique1([1, 2, 3, 4]));  // true
console.log(hasUnique1([1, 2, 2, 3]))  // false

3.3 旋转矩阵

将 N×N 矩阵旋转 90 度。

// 原地旋转 - O(N²)，O(1) 空间
function rotateMatrix(matrix: number[][]): void {
  const n = matrix.length;

  // 按层旋转
  for (let layer = 0; layer < n / 2; layer++) {
    const first = layer;
    const last = n - 1 - layer;

    for (let i = first; i < last; i++) {
      const offset = i - first;

      // 保存上边
      const top = matrix[first][i];

      // 左 → 上
      matrix[first][i] = matrix[last - offset][first];

      // 下 → 左
      matrix[last - offset][first] = matrix[last][last - offset];

      // 右 → 下
      matrix[last][last - offset] = matrix[i][last];

      // 上 → 右
      matrix[i][last] = top;
    }
  }
}

// 测试
const matrix = [
  [1, 2, 3],
  [4, 5, 6],
  [7, 8, 9]
];
rotateMatrix(matrix);
console.log(matrix);
// [7, 4, 1]
// [8, 5, 2]
// [9, 6, 3]

练习 2.1 参考：旋转数组二分搜索

function searchRotated(nums: number[], target: number): number {
  let left = 0, right = nums.length - 1;

  while (left <= right) {
    const mid = Math.floor((left + right) / 2);

    if (nums[mid] === target) return mid;

    // 左边有序
    if (nums[left] <= nums[mid]) {
      if (target >= nums[left] && target < nums[mid]) {
        right = mid - 1;
      } else {
        left = mid + 1;
      }
    } else {
      // 右边有序
      if (target > nums[mid] && target <= nums[right]) {
        left = mid + 1;
      } else {
        right = mid - 1;
      }
    }
  }

  return -1;
}

练习 2.2 参考：归并排序

function mergeSort(arr: number[]): number[] {
  if (arr.length <= 1) return arr;

  const mid = Math.floor(arr.length / 2);
  const left = mergeSort(arr.slice(0, mid));
  const right = mergeSort(arr.slice(mid));

  return merge(left, right);
}

function merge(left: number[], right: number[]): number[] {
  const result: number[] = [];
  let i = 0, j = 0;

  while (i < left.length && j < right.length) {
    if (left[i] <= right[j]) {
      result.push(left[i++]);
    } else {
      result.push(right[j++]);
    }
  }

  return result.concat(left.slice(i)).concat(right.slice(j));
}

练习 2.4 参考：最大子数组和（Kadane 算法）

// 方法1：动态规划 O(n)
function maxSubArray(nums: number[]): number {
  let maxSum = nums[0];
  let currentSum = nums[0];

  for (let i = 1; i < nums.length; i++) {
    currentSum = Math.max(nums[i], currentSum + nums[i]);
    maxSum = Math.max(maxSum, currentSum);
  }

  return maxSum;
}

// 方法2：分治法 O(n log n)
function maxSubArrayDivide(nums: number[]): number {
  function divide(left: number, right: number): number {
    if (left === right) return nums[left];

    const mid = Math.floor((left + right) / 2);
    const leftMax = divide(left, mid);
    const rightMax = divide(mid + 1, right);

    // 跨越中点的最大子数组
    let leftSum = -Infinity, sum = 0;
    for (let i = mid; i >= left; i--) {
      sum += nums[i];
      leftSum = Math.max(leftSum, sum);
    }

    let rightSum = -Infinity; sum = 0;
    for (let i = mid + 1; i <= right; i++) {
      sum += nums[i];
      rightSum = Math.max(rightSum, sum);
    }

    return Math.max(leftMax, rightMax, leftSum + rightSum);
  }

  return divide(0, nums.length - 1);
}

练习 2.5 参考：两数之和

// 方法1：暴力 O(n²)
function twoSum1(nums: number[], target: number): number[] {
  for (let i = 0; i < nums.length; i++) {
    for (let j = i + 1; j < nums.length; j++) {
      if (nums[i] + nums[j] === target) {
        return [i, j];
      }
    }
  }
  return [];
}

// 方法2：哈希表 O(n)
function twoSum2(nums: number[], target: number): number[] {
  const map = new Map<number, number>();

  for (let i = 0; i < nums.length; i++) {
    const complement = target - nums[i];
    if (map.has(complement)) {
      return [map.get(complement)!, i];
    }
    map.set(nums[i], i);
  }

  return [];
}
