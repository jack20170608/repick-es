一、代码可读性

1.1 有意义的命名

// ❌ 糟糕的命名
function c(a, b) {
  let d = 0;
  for (let i = 0; i < a.length; i++) {
    d += a[i];
  }
  return d / a.length;
}

// ✅ 好的命名
function calculateAverage(numbers: number[]): number {
  let sum = 0;
  for (const number of numbers) {
    sum += number;
  }
  return sum / numbers.length;
}

二、错误处理

2.1 防御性编程

// 防御性编程示例
function divide(a: number, b: number): number {
  // 检查输入
  if (typeof a !== 'number' || typeof b !== 'number') {
    throw new TypeError("参数必须是数字");
  }

  // 检查除数
  if (b === 0) {
    throw new Error("除数不能为0");
  }

  return a / b;
}

// 使用 Result 类型（更安全）
type Result<T> = { success: true; value: T } | { success: false; error: Error };

function safeDivide(a: number, b: number): Result<number> {
  if (b === 0) {
    return { success: false, error: new Error("除数不能为0") };
  }
  return { success: true, value: a / b };
}

2.2 错误传播

// 错误应该被正确处理，而不是忽略

// ❌ 错误被吞掉
try {
  doSomething();
} catch (e) {
  // 什么都不做
}

// ✅ 正确处理
try {
  doSomething();
} catch (e) {
  console.error("操作失败:", e);
  throw e;  // 或者返回错误
}

// 或者使用错误边界
function handleError<T>(fn: () => T, fallback: T): T {
  try {
    return fn();
  } catch (e) {
    console.error(e);
    return fallback;
  }
}

三、代码组织

3.1 模块化

// 好的模块组织
// utils/
├── math.ts          # 数学工具
├── string.ts        # 字符串工具
├── array.ts         # 数组工具
└── index.ts         # 导出所有工具

// math.ts
export function average(nums: number[]): number { return nums.reduce((a, b) => a + b, 0) / nums.length; }
export function sum(nums: number[]): number { return nums.reduce((a, b) => a + b, 0); }
export function max(nums: number[]): number { return Math.max(...nums); }
export function min(nums: number[]): number { return Math.min(...nums); }

// index.ts
export * from './math';
export * from './string';
export * from './array';
