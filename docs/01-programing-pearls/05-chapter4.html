三、断言的使用

3.1 防御性编程

// 断言：验证程序状态的假设
function binarySearch<T>(arr: T[], target: T): number {
  // 前置条件
  console.assert(arr.length > 0, "数组不能为空");

  let left = 0;
  let right = arr.length - 1;

  while (left <= right) {
    const mid = Math.floor((left + right) / 2);

    // 不变式：left <= mid <= right
    console.assert(left <= mid && mid <= right, "索引越界");

    if (arr[mid] === target) return mid;
    if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
  }

  // 后置条件
  console.assert(left === right + 1, "搜索结束");

  return -1;
}

3.2 契约式设计

// 契约式设计：前置条件 + 后置条件

// 排序函数的契约
function sort(arr: number[]): number[] {
  // 前置条件：输入是数组
  console.assert(Array.isArray(arr), "输入必须是数组");

  // 核心逻辑
  const result = [...arr].sort((a, b) => a - b);

  // 后置条件1：长度不变
  console.assert(result.length === arr.length, "长度改变");

  // 后置条件2：已排序
  for (let i = 1; i < result.length; i++) {
    console.assert(result[i - 1] <= result[i], "未排序");
  }

  // 后置条件3：是原数组的排列
  console.assert(
    [...arr].sort((a, b) => a - b).every((v, i) => v === result[i]),
    "不是原数组的排列"
  );

  return result;
}

四、测试驱动开发

4.1 TDD 流程

// 红-绿-重构循环

// 1. 红：写一个失败的测试
test('计算阶乘', () => {
  expect(factorial(5)).toBe(120);
});

// 2. 绿：写最简代码让测试通过
function factorial(n) {
  if (n === 0) return 1;
  return n * factorial(n - 1);
}

// 3. 重构：优化代码
function factorial(n, acc = 1) {
  if (n === 0) return acc;
  return factorial(n - 1, n * acc);
}

五、程序正确性验证

5.1 循环不变式

// 二分搜索的循环不变式
function binarySearch(arr, target) {
  let left = 0;
  let right = arr.length - 1;

  // 循环不变式：target 如果在 arr 中，必在 [left, right] 范围内
  while (left <= right) {
    const mid = Math.floor((left + right) / 2);

    if (arr[mid] === target) return mid;
    if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;

    // 每次迭代后，不变式依然成立
  }

  // 循环结束：left > right，target 不在范围内
  return -1;
}

5.2 数学归纳法验证

// 用数学归纳法验证递归函数
// 验证：sum(n) = 1 + 2 + ... + n = n(n+1)/2

// 基例：n = 0
// sum(0) = 0
// 0 * 1 / 2 = 0 ✓

// 归纳假设：假设 sum(k) = k(k+1)/2

// 归纳步骤：证明 sum(k+1) = (k+1)(k+2)/2
// sum(k+1) = sum(k) + (k+1)
//          = k(k+1)/2 + (k+1)
//          = k(k+1)/2 + 2(k+1)/2
//          = (k+2)(k+1)/2
//          = (k+1)(k+2)/2 ✓

function sum(n: number): number {
  if (n === 0) return 0;
  return n + sum(n - 1);
}

// 验证
for (let i = 0; i <= 100; i++) {
  console.assert(
    sum(i) === i * (i + 1) / 2,
    `sum(${i}) 验证失败`
  );
}

七、答案提示

练习 4.1 参考：插入排序验证

function insertionSort(arr: number[]): number[] {
  // 验证输入
  console.assert(Array.isArray(arr), "输入必须是数组");

  const result = [...arr];  // 不修改原数组

  for (let i = 1; i < result.length; i++) {
    const key = result[i];
    let j = i - 1;

    // 循环不变式：result[0...i-1] 已排序
    for (let k = 0; k < i; k++) {
      console.assert(
        k === 0 || result[k - 1] <= result[k],
        "数组未排序"
      );
    }

    while (j >= 0 && result[j] > key) {
      result[j + 1] = result[j];
      j--;
    }
    result[j + 1] = key;
  }

  // 验证输出：已排序
  for (let i = 1; i < result.length; i++) {
    console.assert(result[i - 1] <= result[i], "结果未排序");
  }

  // 验证输出：是原数组的排列
  console.assert(
    result.sort((a, b) => a - b).every((v, i) => v === [...arr].sort((a, b) => a - b)[i]),
    "不是原数组的排列"
  );

  return result;
}

function testReverse() {
