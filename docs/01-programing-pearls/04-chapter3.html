二、基本数据结构对比

数据结构 访问 查找 插入 删除 使用场景
数组 O(1) O(n) O(n) O(n) 需要快速随机访问
有序数组 O(1) O(log n) O(n) O(n) 查找多，修改少
链表 O(n) O(n) O(1) O(1)* 频繁插入删除
哈希表 - O(1) O(1) O(1) 快速查找
二叉搜索树 O(log n) O(log n) O(log n) O(log n) 有序遍历 + 查找
堆 - O(1) 极值 O(log n) O(log n) 优先级队列

3.1 LRU 缓存

需要快速获取和删除最近最少使用的数据。

// LRU 缓存：哈希表 + 双向链表
class LRUCache {
  private capacity: number;
  private cache: Map<number, number>;

  constructor(capacity: number) {
    this.capacity = capacity;
    this.cache = new Map();
  }

  get(key: number): number | -1 {
    if (!this.cache.has(key)) return -1;

    // 移到末尾（最近使用）
    const value = this.cache.get(key)!;
    this.cache.delete(key);
    this.cache.set(key, value);

    return value;
  }

  put(key: number, value: number): void {
    if (this.cache.has(key)) {
      this.cache.delete(key);
    } else if (this.cache.size >= this.capacity) {
      // 删除最老的（第一个）
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }
    this.cache.set(key, value);
  }
}

3.2 频率统计

// 统计出现频率最高的 K 个元素
function topKFrequent(nums: number[], k: number): number[] {
  // 1. 统计频率
  const freq: Map<number, number> = new Map();
  for (const num of nums) {
    freq.set(num, (freq.get(num) || 0) + 1);
  }

  // 2. 按频率分组
  const buckets: number[][] = [];
  for (let i = 0; i <= nums.length; i++) {
    buckets[i] = [];
  }
  for (const [num, count] of freq) {
    buckets[count].push(num);
  }

  // 3. 从高频到低频收集
  const result: number[] = [];
  for (let i = buckets.length - 1; i >= 0 && result.length < k; i--) {
    result.push(...buckets[i]);
  }

  return result;
}

console.topKFrequent([1,1,1,2,2,3], 2);  // [1, 2]

3.3 栈结构应用

// 有效的括号
function isValid(s: string): boolean {
  const stack: string[] = [];
  const map: Record<string, string> = {
    '}': '{',
    ']': '[',
    ')': '('
  };

  for (const c of s) {
    if ('({['.includes(c)) {
      stack.push(c);
    } else if (')}]'.includes(c)) {
      if (stack.length === 0 || stack.pop() !== map[c]) {
        return false;
      }
    }
  }

  return stack.length === 0;
}

// 表达式求值
function evalRPN(tokens: string[]): number {
  const stack: number[] = [];
  const ops: Record<string, (a: number, b: number) => number> = {
    '+': (a, b) => a + b,
    '-': (a, b) => a - b,
    '*': (a, b) => a * b,
    '/': (a, b) => Math.trunc(a / b)
  };

  for (const token of tokens) {
    if (ops[token]) {
      const b = stack.pop()!;
      const a = stack.pop()!;
      stack.push(ops[token](a, b));
    } else {
      stack.push(Number(token));
    }
  }

  return stack[0];
}

console.evalRPN(["2", "1", "+", "3", "*"]);  // 9

3.4 队列结构应用

// 滑动窗口最大值
function maxSlidingWindow(nums: number[], k: number): number[] {
  const result: number[] = [];
  const deque: number[] = [];  // 存储索引

  for (let i = 0; i < nums.length; i++) {
    // 移除超出窗口的元素
    if (deque.length > 0 && deque[0] <= i - k) {
      deque.shift();
    }

    // 移除比当前元素小的元素
    while (deque.length > 0 && nums[deque[deque.length - 1]] < nums[i]) {
      deque.pop();
    }

    deque.push(i);

    // 记录窗口最大值
    if (i >= k - 1) {
      result.push(nums[deque[0]]);
    }
  }

  return result;
}

console.maxSlidingWindow([1,3,-1,-3,5,3,6,7], 3);
// [3, 3, 5, 5, 6, 7]

class PhoneBook {
  nameToPhone: Map<string, string>;
  phoneToName: Map<string, string>;

  constructor() {
    this.nameToPhone = new Map();
    this.phoneToName = new Map();
  }

  add(name: string, phone: string): void {
    this.nameToPhone.set(name, phone);
    this.phoneToName.set(phone, name);
  }

    return this.nameToPhone.get(name);
  }

  findByPhone(phone: string): string | undefined {
    return this.phoneToName.get(phone);
  }
}

4.2 案例：文本编辑器撤销

// 撤销功能：需要记录历史状态
class TextEditor {
  private text: string = "";
  private undoStack: string[] = [];
  private redoStack: string[] = [];

  type(c: string): void {
    this.undoStack.push(this.text);
    this.text += c;
    this.redoStack = [];  // 新操作清除 redo
  }

  undo(): void {
    if (this.undoStack.length === 0) return;
    this.redoStack.push(this.text);
    this.text = this.undoStack.pop()!;
  }

  redo(): void {
    if (this.redoStack.length === 0) return;
    this.undoStack.push(this.text);
    this.text = this.redoStack.pop()!;
  }

  getText(): string {
    return this.text;
  }
}

// 思路：两个堆 - 最大堆 + 最小堆
// 最大堆：存储较小的一半
// 最小堆：存储较大的一半

class MedianFinder {
  private low: number[] = [];   // 最大堆（用负数模拟）
  private high: number[] = [];  // 最小堆

  addNum(num: number): void {
    // 插入到 low（最大堆）
    this.low.push(-num);
    siftUp(this.low);
    siftDown(this.low, this.low.length - 1);

    // 平衡：low 顶部移到 high
    const max = -pop(this.low);
    this.high.push(max);
    siftUp(this.high);

    // 确保 low >= high 且 low <= high + 1
    if (this.low.length < this.high.length) {
      const min = pop(this.high);
      this.low.push(-min);
      siftUp(this.low);
    }
  }

  findMedian(): number {
    if (this.low.length > this.high.length) {
      return -this.low[0];
    }
    return (-this.low[0] + this.high[0]) / 2;
  }

  private siftUp(heap: number[]): void { /* 上浮操作 */ }
  private siftDown(heap: number[], i: number): void { /* 下沉操作 */ }
  private pop(heap: number[]): number {
    const top = heap[0];
    heap[0] = heap.pop()!;
    siftDown(heap, 0);
    return top;
  }
}

练习 3.2 参考：栈实现队列

// 两个栈实现队列
class QueueWithStacks {
  private inStack: number[] = [];
  private outStack: number[] = [];

  push(x: number): void {
    this.inStack.push(x);
  }

  pop(): number {
    if (this.outStack.length === 0) {
      while (this.inStack.length > 0) {
        this.outStack.push(this.inStack.pop()!);
      }
    }
    return this.outStack.pop()!;
  }

  peek(): number {
    if (this.outStack.length === 0) {
      while (this.inStack.length > 0) {
        this.outStack.push(this.inStack.pop()!);
      }
    }
    return this.outStack[this.outStack.length - 1];
  }

  empty(): boolean {
    return this.inStack.length === 0 && this.outStack.length === 0;
  }
}
