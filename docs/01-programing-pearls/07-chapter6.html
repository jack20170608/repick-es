一、Amdahl 定律

系统性能提升受限于并行化部分的比例：

Speedup = 1 / (S + (1-S)/N)

其中：

  • S = 可并行化部分的比例
  • N = 处理器/核心数
// Amdahl 定律示例
function calculateSpeedup(serialFraction: number, cores: number): number {
  const S = serialFraction;
  const N = cores;
  return 1 / (S + (1 - S) / N);
}

// 50% 可并行化
console.log(calculateSpeedup(0.5, 4));  // 1.6x 加速

// 90% 可并行化
console.log(calculateSpeedup(0.1, 4));  // 3.3x 加速

// 10% 串行化
console.log(calculateSpeedup(0.01, 4));  // 3.9x 加速

// 结论：即使无限多核，10% 串行化限制最大加速到 10x

二、性能测量方法

2.1 基准测试

// 精确的基准测试
function benchmark(
  fn: () => void,
  iterations: number = 1000
): number {
  // 预热
  for (let i = 0; i < 10; i++) fn();

  // 实际测量
  const start = performance.now();
  for (let i = 0; i < iterations; i++) fn();
  const end = performance.now();

  return (end - start) / iterations;
}

// 示例：比较两种排序
const arr = Array.from({ length: 1000 }, () => Math.random());

const quickSortTime = benchmark(() => {
  [...arr].sort((a, b) => a - b);
});

const bubbleSortTime = benchmark(() => {
  const copy = [...arr];
  for (let i = 0; i < copy.length; i++) {
    for (let j = 0; j < copy.length - i - 1; j++) {
      if (copy[j] > copy[j + 1]) {
        [copy[j], copy[j + 1]] = [copy[j + 1], copy[j]];
      }
    }
  }
});

console.log(`快速排序: ${quickSortTime.toFixed(3)}ms`);
console.log(`冒泡排序: ${bubbleSortTime.toFixed(3)}ms`);
console.log(`速度比: ${(bubbleSortTime / quickSortTime).toFixed(1)}x`);

function profileSection(label: string, fn: () => void): void {
  const start = process.hrtime.bigint();
  fn();
  const end = process.hrtime.bigint();
  const ms = Number(end - start) / 1000000;
  console.log(`${label}: ${ms.toFixed(2)}ms`);
}

profileSection("数据加载", () => loadData());
profileSection("数据处理", () => processData());
profileSection("UI 渲染", () => renderUI());

四、优化策略

4.1 优化优先级

// 优化优先级
// 1. 算法优化 - 最高回报
//    O(n²) → O(n log n) 可能提升 1000 倍

// 2. 数据结构优化 - 高回报
//    数组 → 哈希表 可能提升 100 倍

// 3. 代码优化 - 中等回报
//    循环展开、减少函数调用可能提升 2-5 倍

// 4. 编译器优化 - 低回报
//    开启优化选项通常提升 10-20%

// 示例：算法优化 vs 代码优化
// ❌ 代码优化（从小处着手）
function sum1(n: number): number {
  let result = 0;
  for (let i = 1; i <= n; i++) {
    result += i;
  }
  return result;
}

// ✅ 算法优化（从大处着眼）
function sum2(n: number): number {
  return n * (n + 1) / 2;  // 数学公式
}

4.2 80-20 法则

// 80% 的时间花在 20% 的代码上
// 找出热点，重点优化

// 示例：日志系统优化
class Logger {
  private buffer: string[] = [];
  private flushInterval: number;

  constructor() {
    this.flushInterval = setInterval(() => this.flush(), 1000);
  }

  log(msg: string): void {
    // 批量写入，避免频繁 IO
    this.buffer.push(msg);
  }

  private flush(): void {
    if (this.buffer.length > 0) {
      console.log(`[${this.buffer.length} 条日志]`);
      this.buffer = [];
    }
  }
}
