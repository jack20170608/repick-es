一、书籍简介

《编程珠玑》是计算机科学领域的经典著作，由贝尔实验室的 Jon Bentley 编写。

本书不是一本算法教科书，而是一本关于编程思考方式的书。它强调：

  • 问题分析 - 在动手写代码之前，充分理解问题
  • 算法选择 - 正确的算法比硬件更重要
  • 代码质量 - 简洁、优雅、可读的代码
  • 性能优化 - 找到真正的瓶颈，而不是过早优化

核心哲学

"正确的程序还不够，还要写出高效的程序。"
"编程的本质是思考，而不是打字。"

4.1 时间复杂度

// 常见时间复杂度（从快到慢）
O(1)        // 常数时间 - 数组随机访问
O(log n)    // 对数时间 - 二分搜索
O(n)        // 线性时间 - 遍历
O(n log n)  // 线性对数时间 - 快速排序
O(n²)       // 平方时间 - 冒泡排序
O(2ⁿ)       // 指数时间 - 递归斐波那契
O(n!)       // 阶乘时间 - 全排列

4.2 空间复杂度

// 空间复杂度分析
// 原地算法：O(1) 额外空间
// 递归算法：注意调用栈深度

// 示例：反转数组
function reverse(arr: number[]): number[] {
  const result: number[] = [];  // O(n) 空间
  for (let i = arr.length - 1; i >= 0; i--) {
    result.push(arr[i]);
  }
  return result;
}

// 原地版本：O(1) 空间
function reverseInPlace(arr: number[]): void {
  let left = 0;
  let right = arr.length - 1;
  while (left < right) {
    [arr[left], arr[right]] = [arr[right], arr[left]];
    left++;
    right--;
  }
}

4.3 二分搜索

// 标准二分搜索
function binarySearch<T>(arr: T[], target: T, compare: (a: T, b: T) => number): number {
  let left = 0;
  let right = arr.length - 1;

  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    const cmp = compare(arr[mid], target);

    if (cmp === 0) {
      return mid;  // 找到
    } else if (cmp < 0) {
      left = mid + 1;
    } else {
      right = mid - 1;
    }
  }

  return -1;  // 未找到
}

// 变体：查找第一个/最后一个
function findFirst<T>(arr: T[], target: T): number {
  let left = 0, right = arr.length - 1, result = -1;

  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    if (arr[mid] === target) {
      result = mid;
      right = mid - 1;  // 继续向左找
    } else if (arr[mid] < target) {
      left = mid + 1;
    } else {
      right = mid - 1;
    }
  }

  return result;
}

4.4 散列（Hash）

// 简单的哈希表实现
class HashMap<K, V> {
  private buckets: Map<number, [K, V][]> = new Map();

  hash(key: K): number {
    const str = String(key);
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      hash = ((hash << 5) - hash) + str.charCodeAt(i);
      hash = hash & hash;  // 防止溢出
    }
    return Math.abs(hash);
  }

  set(key: K, value: V): void {
    const h = hash(key);
    const bucket = this.buckets.get(h) || [];
    for (let i = 0; i < bucket.length; i++) {
      if (bucket[i][0] === key) {
        bucket[i][1] = value;
        return;
      }
    }
    bucket.push([key, value]);
    this.buckets.set(h, bucket);
  }

  get(key: K): V | undefined {
    const h = hash(key);
    const bucket = this.buckets.get(h);
    if (!bucket) return undefined;
    for (const [k, v] of bucket) {
      if (k === key) return v;
    }
    return undefined;
  }
}

4.5 堆排序

// 堆排序
function heapSort(arr: number[]): number[] {
  const n = arr.length;

  // 构建最大堆
  for (let i = Math.floor(n / 2) - 1; i >= 0; i--) {
    siftDown(arr, n, i);
  }

  // 提取元素
  for (let i = n - 1; i > 0; i--) {
    [arr[0], arr[i]] = [arr[i], arr[0]];
    siftDown(arr, i, 0);
  }

  return arr;
}

function siftDown(arr: number[], n: number, i: number): void {
  while (true) {
    let largest = i;
    const left = 2 * i + 1;
    const right = 2 * i + 2;

    if (left < n && arr[left] > arr[largest]) {
      largest = left;
    }
    if (right < n && arr[right] > arr[largest]) {
      largest = right;
    }

    if (largest === i) break;

    [arr[i], arr[largest]] = [arr[largest], arr[i]];
    i = largest;
  }
}

// 测试
console.log(heapSort([3, 1, 4, 1, 5, 9, 2, 6]));
// [1, 1, 2, 3, 4, 5, 6, 9]

五、学习建议

5.1 学习路径

  1. 第一遍通读：快速浏览，了解整体结构
  2. 第二遍精读：跟着书中的例子动手实现
  3. 第三遍实践：用学到的思想解决实际问题

5.2 配套练习

  • 实现书中提到的所有算法
  • 分析自己代码的时间/空间复杂度
  • 尝试用不同方法解决同一问题

5.3 延伸阅读

  • 《算法导论》- Thomas H. Cormen
  • 《代码整洁之道》- Robert C. Martin
  • 《算法》- Robert Sedgewick

六、总结

维度 要点
核心理念 思考优先于编码
算法重点 二分、排序、哈希、堆
性能优化 先找瓶颈，再优化
代码质量 简洁、可读、可测试

《编程珠玑》不是一本读一遍就扔的书。它值得反复阅读，每次都会有新的收获。
